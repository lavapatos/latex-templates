    \item ¿Cuál es el resultado de $-3\cdot(4-7)+2$?
    \begin{enumerate}
        \item $-11$
        \item $-7$
        \item $11$ % <- correcta
        \item $5$
    \end{enumerate}

    \item Un polerón tiene un precio de \$12.000 y se vende con un descuento del $15\%$.

    ¿Cuál de las siguientes expresiones representa el precio con descuento, en pesos?
    \begin{enumerate}
        \item $0.15\cdot 12{.}000$
        \item $0.85\cdot 12{.}000$ % <- correcta
        \item $12{.}000-15$
        \item $1.15\cdot 12{.}000$
    \end{enumerate}
